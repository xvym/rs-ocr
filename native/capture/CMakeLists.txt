cmake_minimum_required(VERSION 3.28)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(rscapture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "rscapture requires Windows x64 and MSVC")
endif()
find_package(JNI REQUIRED COMPONENTS JVM)
add_library(rscapture SHARED capture.cpp)
target_compile_features(rscapture PRIVATE cxx_std_20)
target_include_directories(rscapture PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_definitions(rscapture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(rscapture PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(rscapture PRIVATE d3d11 dxgi windowsapp dwmapi)
